<script>
var D=__DATA__;
(function(){
"use strict";
var IM={};D.images.forEach(function(i){IM[i.id]=i});
var saved=D.configs||{};
var dates=Object.keys(saved).sort();
function pad(n){return String(n).padStart(2,"0")}
function isoOf(d){return d.getFullYear()+"-"+pad(d.getMonth()+1)+"-"+pad(d.getDate())}
function parse(s){var p=s.split("-").map(Number);return new Date(p[0],p[1]-1,p[2],12)}
function cap(s){return s.charAt(0).toUpperCase()+s.slice(1)}
function ink(c){var n=parseInt(c.slice(1),16),r=n>>16,g=(n>>8)&255,b=n&255;return (0.299*r+0.587*g+0.114*b)>150?"#0b0b0b":"#ffffff"}
function el(tag,cls,txt){var e=document.createElement(tag);if(cls)e.className=cls;if(txt!=null)e.textContent=txt;return e}

function initial(){
  var h=(location.hash||"").slice(1);
  if(saved[h])return h;
  var t=isoOf(new Date());
  if(saved[t])return t;
  var past=dates.filter(function(d){return d<=t});
  if(past.length)return past[past.length-1];
  return dates[0]||null;
}
var cur=initial();
var view=parse(cur||isoOf(new Date()));view.setDate(1);

function showDay(){
  var box=document.getElementById("cards");box.textContent="";
  if(!cur){
    document.getElementById("date").hidden=true;
    box.appendChild(el("p","empty","Todavía no hay configuraciones guardadas."));
    return;
  }
  var d=parse(cur),c=saved[cur];
  document.getElementById("dWeek").textContent=d.toLocaleDateString("es-ES",{weekday:"long"});
  document.getElementById("dDay").textContent=d.getDate();
  document.getElementById("dMonth").textContent=cap(d.toLocaleDateString("es-ES",{month:"long"}))+" "+d.getFullYear();
  var order=(c.order||[]).filter(function(id){return IM[id]});
  D.images.forEach(function(i){if(order.indexOf(i.id)<0)order.push(i.id)});
  order.forEach(function(id){
    var im=IM[id],card=el("article","card"),frame=el("div","frame");
    var img=el("img");img.src=im.src;img.alt=im.alt;img.width=im.w;img.height=im.h;img.loading="lazy";
    frame.appendChild(img);
    (c.placed||[]).filter(function(p){return p.img===id}).forEach(function(p){
      var t=el("div","tag",p.name),col=/^#[0-9a-f]{6}$/i.test(p.c||"")?p.c:"#0b0b0b";
      t.style.left=p.x+"%";t.style.top=p.y+"%";t.style.setProperty("--s",p.s);
      t.style.setProperty("--c",col);t.style.setProperty("--ci",ink(col));
      frame.appendChild(t);
    });
    card.appendChild(frame);box.appendChild(card);
  });
}

function drawCal(){
  var root=document.getElementById("cal");root.textContent="";
  var top=el("div","top"),prev=el("button","nav","‹"),next=el("button","nav","›");
  prev.type=next.type="button";prev.setAttribute("aria-label","Mes anterior");next.setAttribute("aria-label","Mes siguiente");
  prev.onclick=function(){view.setMonth(view.getMonth()-1);drawCal()};
  next.onclick=function(){view.setMonth(view.getMonth()+1);drawCal()};
  top.appendChild(prev);top.appendChild(el("span","mo",view.toLocaleDateString("es-ES",{month:"long",year:"numeric"})));top.appendChild(next);
  var grid=el("div","grid");
  ["L","M","X","J","V","S","D"].forEach(function(w){grid.appendChild(el("span","wd",w))});
  var first=(view.getDay()+6)%7,n=new Date(view.getFullYear(),view.getMonth()+1,0).getDate(),today=isoOf(new Date());
  for(var i=0;i<first;i++)grid.appendChild(el("span","day blank"));
  for(var d=1;d<=n;d++){
    (function(d){
      var key=view.getFullYear()+"-"+pad(view.getMonth()+1)+"-"+pad(d),has=!!saved[key];
      var b=el("button","day"+(key===today?" today":"")+(has?" saved":"")+(key===cur?" sel":""),String(d));
      b.type="button";
      b.setAttribute("aria-label",parse(key).toLocaleDateString("es-ES",{weekday:"long",day:"numeric",month:"long",year:"numeric"}));
      if(!has)b.disabled=true;
      else b.onclick=function(){cur=key;history.replaceState(null,"","#"+key);showDay();drawCal()};
      grid.appendChild(b);
    })(d);
  }
  root.appendChild(top);root.appendChild(grid);
}
window.addEventListener("hashchange",function(){var h=location.hash.slice(1);if(saved[h]&&h!==cur){cur=h;view=parse(h);view.setDate(1);showDay();drawCal()}});
if(D.generated){
  var g=new Date(D.generated);
  document.getElementById("upd").textContent="Datos actualizados el "+g.toLocaleDateString("es-ES",{day:"numeric",month:"long",year:"numeric"})+".";
}
showDay();drawCal();
})();
